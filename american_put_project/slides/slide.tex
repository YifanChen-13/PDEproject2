\section{Model}

\begin{frame}[t]{From optimal stopping to the pricing operator}
\small
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
The payoff is $\psi(S)=(K-S)^+$. The holder chooses when to exercise.
\[
 V(S)=\sup_{\tau\ge0}\E_S^{\mathbb Q}\!\left[e^{-r\tau}\psi(S_\tau)\right]
 \ge \psi(S).
\]
Under the risk-neutral measure $\mathbb Q$, $dS_t=\mu\,dt+\nu\,dW_t^{\mathbb Q}$,
with $\mu=(r-q)S_t$ and $\nu=\sigma(S_t)S_t$.
\par
Where $V$ is smooth, apply Itô's formula, then the product rule with $e^{-rt}$.
\[
\begin{aligned}
 dV(S_t)
  &= V'(S_t)\,dS_t+\tfrac12\nu^2 V''(S_t)\,dt,\\[.4em]
 d\big(e^{-rt}V(S_t)\big)
  &=e^{-rt}\bigl[
      \underbrace{\big(\tfrac12\nu^2 V''+\mu V'-rV\big)}_{\Lop V(S_t)}
        \,dt+\nu V'\,dW_t^{\mathbb Q}
    \bigr].
\end{aligned}
\]
\begin{block}{What $\Lop V$ measures}
The Brownian term has zero conditional mean. $\Lop V(S)$ is the local expected change per unit time in discounted option value.
\end{block}
\end{frame}

\begin{frame}[t]{Why continuation gives $\Lop V=0$}
\small
Inside the continuation region, take a short step $\theta$, ending earlier if the stock
leaves a small neighbourhood inside that region.
Waiting until $\theta$ and then acting optimally gives
\[
 V(S)=\E_S^{\mathbb Q}\!\left[e^{-r\theta}V(S_\theta)\right]
 \quad\Longrightarrow\quad \Lop V(S)=0.
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Continue}
$V>\psi$ and $\Lop V=0$. The short step is already part of the optimal strategy.
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Exercise}
$V=\psi=K-S$, so $\Lop V=qS-rK<0$ in the exercise interior. At $S=60$ in Stage 2, this rate is $-3.8$.
\end{block}
\end{column}
\end{columns}
\[
 \boxed{\min\{-\Lop V,\ V-\psi\}=0}
 \qquad
 V(b)=K-b,\quad V'(b)=-1.
\]
\end{frame}

\begin{frame}[t]{Two stages and the exact benchmark}
\small
Both stages use strike $K=100$ and risk-free rate $r=0.05$.

\begin{columns}[T,onlytextwidth]
\begin{column}{.60\textwidth}
\begin{block}{Stage 1 exercise boundary}
\small
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
For $\sigma=0.20$ and $q=0$, insert $V=CS^\lambda$ into $\Lop V=0$.
\[
 0.02\lambda(\lambda-1)+0.05\lambda-0.05=0.
\]
The roots are $1$ and $-5/2$. Decay at infinity selects $V=CS^{-5/2}$.
Match value and slope at $b$.
\[
\begin{gathered}
 V(b)=Cb^{-5/2}=100-b,\\
 V'(b)=-\tfrac{5}{2b}V(b)=-1,\\[.4em]
 b=\tfrac52(100-b)\quad\Longrightarrow\quad
 \boxed{b=\tfrac{500}{7}\approx71.4286}.
\end{gathered}
\]
\end{block}
\end{column}
\begin{column}{.37\textwidth}
\begin{block}{Stage 2 local volatility}
\small
The dividend yield is $q=0.02$.
\[
\begin{aligned}
 \sigma(S)&=0.20[1+0.35\tanh z],\\
 z&=\frac{\log(S/100)}{0.40}.
\end{aligned}
\]
On $[0,L]$, impose $V(0)=100$ and $V(L)=0$.
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Numerical methods}

\begin{frame}[t]{Finite differences give a three-point equation}
\small
Use nodes $S_i=ih$ and unknowns $v_i\approx V(S_i)$. At each interior node, set
$d_i=\tfrac12\sigma(S_i)^2 S_i^2$ and $\mu_i=(r-q)S_i$.
\[
-\Lop_h v_i
 =-d_i\,\underbrace{\frac{v_{i-1}-2v_i+v_{i+1}}{h^2}}_{\text{centred diffusion}}
  -\mu_i\,\underbrace{\frac{v_{i+1}-v_i}{h}}_{\text{forward drift}}
  +rv_i.
\]
Collect the coefficients of the left neighbour, current node and right neighbour.
\[
-\Lop_h v_i=\ell_i v_{i-1}+c_i v_i+u_i v_{i+1},
\]
\[
 \ell_i=-\frac{d_i}{h^2},\qquad
 c_i=\frac{2d_i}{h^2}+\frac{\mu_i}{h}+r,\qquad
 u_i=-\frac{d_i}{h^2}-\frac{\mu_i}{h}.
\]
A continuation node sets this expression to zero. An exercise node sets $v_i=\psi(S_i)$.
\end{frame}

\begin{frame}[t]{From three-point equations to the matrix}
\small
Take Stage 2 with $L=200$ and $h=50$. The three unknowns are at $S=50,100,150$.
Stacking the continuation equations gives $Av=f$ with the coefficients rounded here.
\[
 \underbrace{\begin{bmatrix}
 0.09802 & -0.03901 & 0\\
 -0.08 & 0.27 & -0.14\\
 0 & -0.28966 & 0.71931
 \end{bmatrix}}_{A}
 \underbrace{\begin{bmatrix}v_1\\v_2\\v_3\end{bmatrix}}_{v}
 =
 \underbrace{\begin{bmatrix}0.90105\\0\\0\end{bmatrix}}_{f}.
\]
The endpoint terms move to $f_1=-\ell_1v_0$ and $f_3=-u_3v_4$,
where $v_0=100$ and $v_4=0$.
\begin{block}{Allowing exercise gives a linear complementarity problem}
For each node, compare the continuation residual $a_i=(Av-f)_i$
with the payoff gap $g_i=v_i-\psi(S_i)$.
\[
 a_i\ge0,\qquad g_i\ge0,\qquad \min(a_i,g_i)=0.
\]
\end{block}
\end{frame}

\begin{frame}[t]{Solving the tridiagonal system in code}
\small
Fix the policy, then solve $Bv=\mathrm{rhs}$ by elimination and back substitution.

\begin{columns}[T,onlytextwidth]
\begin{column}{.43\textwidth}
\begin{block}{1. Assemble the selected rows}
\small
Exercise at $S=50$, giving $v_1=50$. Keep the two continuation rows.
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
\setlength{\arraycolsep}{3pt}
\[
 \left[\begin{array}{@{}rrr|r@{}}
 1 & 0 & 0 & 50\\
 -0.08 & 0.27 & -0.14 & 0\\
 0 & -0.28966 & 0.71931 & 0
 \end{array}\right].
\]
\footnotesize
Rightmost column is $\mathrm{rhs}$. Entries are rounded.
\end{block}
\end{column}
\begin{column}{.54\textwidth}
\begin{block}{2. Pivot and eliminate}
\small
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
Use row 1 to cancel $v_1$ in row 2.
\[
 R_2\gets R_2+0.08R_1
 \quad\Longrightarrow\quad 0.27v_2-0.14v_3=4.
\]
Swap rows 2 and 3 since $|-0.28966|>0.27$.
Use the new row 2 to cancel $v_2$ in row 3.
\[
 R_3\gets R_3-\tfrac{0.27}{-0.28966}R_2
 \quad\Longrightarrow\quad 0.53050v_3\approx4.
\]
\end{block}
\end{column}
\end{columns}

\begin{block}{3. Back-substitute from the last row}
\small
$v_3\approx 4/0.53050\approx7.54007,\qquad
v_2\approx (0.71931/0.28966)v_3\approx18.72448,\qquad
v_1=50.$
\end{block}

\footnotesize
\texttt{solve\_banded((1, 1), ab, rhs)} uses LAPACK \texttt{DGTSV} with the three diagonals stored in \texttt{ab}.
\end{frame}

\begin{frame}[t]{Policy iteration and projected SOR}
\small
Both solvers start from the immediate payoff, $v^{(0)}=\psi_h$.
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{Policy iteration}
\small
\begin{enumerate}
\item Compute $a$ and $g$ from the current vector.
\item Where $g_i<a_i$, use $v_i=\psi_i$.
      Otherwise retain $(Av)_i=f_i$.
\item Solve the selected tridiagonal system.
\end{enumerate}
\end{block}
\end{column}
\begin{column}{.48\textwidth}
\begin{block}{Projected SOR}
\small
Sweep left to right. Solve one continuation row with the current
neighbours to obtain a trial value $p_i$. Update
\[
v_i^{\mathrm{new}}=\max\{\psi_i,\,
    v_i^{\mathrm{old}}+\omega(p_i-v_i^{\mathrm{old}})\}.
\]
The next row uses the updated value.
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}[t]{A continuous check using shooting}
\small
Stage 2 on $[0,600]$. Find where the continuation curve joins the payoff $K-S$.

\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\begin{block}{1. Integrate the ODE with RK45}
\footnotesize
Set $y_1=W$, $y_2=W'$, $d=\sigma(S)^2S^2/2$ and $\mu=(r-q)S$.
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
\[
 \begin{aligned}
 y_1'&=y_2, & y_2'&=(r y_1-\mu y_2)/d,\\
 y_1(600)&=0, & y_2(600)&=-1.
 \end{aligned}
\]
\texttt{solve\_ivp} integrates from $S=600$ down to $10$.
RK45 samples several slopes per step. It uses the fifth-order update and compares it with the fourth-order estimate to adapt the step size.
\par
Dense output gives $W$ and $W'$ at each trial $b$.
\end{block}
\end{column}
\begin{column}{.49\textwidth}
\begin{block}{2. Scale and match at the boundary}
\footnotesize
The ODE is linear and homogeneous.
\setlength{\abovedisplayskip}{4pt}
\setlength{\belowdisplayskip}{4pt}
\setlength{\abovedisplayshortskip}{3pt}
\setlength{\belowdisplayshortskip}{3pt}
\[
 \Lop(cW)=c\Lop W=0,\qquad cW(600)=0.
\]
The initial slope $-1$ sets a scale for $W$.
At $b$, match the payoff's slope and value.
\[
 cW'(b)=-1,\qquad cW(b)=K-b.
\]
With $c=-1/W'(b)$, \texttt{brentq} solves
\[
 W(b)+(K-b)W'(b)=0\quad (30<b<99).
\]
\end{block}
\end{column}
\end{columns}

\smallskip
\textbf{Result.} $b\approx69.26448$. For $S>b$, $V(S)=(K-b)W(S)/W(b)$, giving $V(100)\approx16.06008$.
\par
Adaptive collocation confirms both results on the same domain.
\end{frame}

\section{Conclusions}

\begin{frame}[t]{Conclusions and limits}
\begin{itemize}
\item \textbf{Validation.} Stage 1 prices approach the exact benchmark under mesh refinement.
\item \textbf{Convergence.} Stage 2 approaches the same-domain continuous solution at approximately first order.
